def test_m2m_total_leak(env):
    client, as_user, admin, reader, group_id, _S = env
    rid = _relation(client, name="co", cardinality="many_to_many", childKey=None,
        linkCollectionId="rel_liens", linkParentKey="parcelle_id", linkChildKey="proprio_id").json()["id"]
    pid = client.post("/v1/collections/rel_parcelles/items", json=_f({"ref":"A"})).json()["id"]
    for nom, x in (("in",5.0),("out",50.0),("out2",60.0)):
        o = client.post("/v1/collections/rel_proprios/items", json=_f({"nom":nom},x=x,y=x)).json()["id"]
        client.post("/v1/collections/rel_liens/items", json=_f({"parcelle_id":pid,"proprio_id":o}))
    client.put("/v1/collections/rel_liens/sharing", json={"public": True, "groups": []})
    client.put(f"/v1/collections/rel_proprios/geo-limits/group/{group_id}", json={"geometry":SQUARE})
    as_user(reader)
    rr = _related(client, pid, rid); print("M2M status", rr.status_code, rr.text[:300]); b = rr.json()
    print("M2M", [f["properties"]["nom"] for f in b["features"]], b["numberReturned"], b["numberMatched"])

def test_oneone_oracle_and_restrict(env):
    client, as_user, admin, reader, group_id, _S = env
    _relation(client, cardinality="one_to_one")
    pid = client.post("/v1/collections/rel_parcelles/items", json=_f({"ref":"A"})).json()["id"]
    client.post("/v1/collections/rel_proprios/items", json=_f({"parcelle_id":pid},x=50,y=50))
    client.put(f"/v1/collections/rel_proprios/geo-limits/group/{group_id}", json={"geometry":SQUARE})
    client.put("/v1/collections/rel_proprios/sharing", json={"public": True, "groups":[{"groupId":group_id,"role":"editor"}]})
    as_user(reader)
    r = client.post("/v1/collections/rel_proprios/items", json=_f({"parcelle_id":pid}))
    print("1-1 oracle", r.status_code, r.text[:200])
